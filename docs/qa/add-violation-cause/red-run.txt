---------------------------------------------------------------------------------------------------------------
// Bounded search for the smoke board p6 (tasks.md 3.6 (b)): a valid capture value (unique-solution givens of an engine puzzle, plus WRONG
// entries on non-given cells) on which the board is rule-clean and the page's first hint (hint(board, 4)) is a fill that breaks a rule.
import { countSolutions, findViolations, generate, hint } from '/Users/vecktor/projects/fwdays/exam/binarka/.claude/worktrees/trusting-liskov-878a97/src/engine/index';
type C = 0 | 1 | null;
const SEEDS = Number(process.argv[2] ?? 300);
const TRIES = Number(process.argv[3] ?? 300);
let rng = 12345;
const rand = (): number => { rng = (Math.imul(rng, 1103515245) + 12345) >>> 0; return rng / 4294967296; };
let examined = 0;
for (const size of [6]) for (let seed = 1; seed <= SEEDS; seed++) {
  const p = generate(size, seed, 1);
  const g = p.givens as C[][];
  if (countSolutions(g) !== 1) continue;
  for (let t = 0; t < TRIES; t++) {
    examined++;
    const k = 1 + Math.floor(rand() * 8);
    const e: C[][] = g.map((row) => row.map(() => null));
    for (let i = 0; i < k; i++) {
      const r = Math.floor(rand() * size), c = Math.floor(rand() * size);
      if (g[r][c] !== null) continue;
      e[r][c] = (1 - (p.solution[r][c] as number)) as 0 | 1; // a wrong entry
    }
    const b = g.map((row, r) => row.map((v, c) => v ?? e[r][c]));
    if (findViolations(b).length > 0) continue;
    const h = hint(b, 4);
    if (h.kind !== 'fill') continue;
    const after = b.map((row) => [...row]);
    after[h.row][h.col] = h.value;
    const v = findViolations(after);
    if (v.length === 0) continue;
    console.log(JSON.stringify({ found: true, size, seed, level: 1, examined, entries: e, givens: g, hint: { row: h.row, col: h.col, value: h.value, rule: (h as { rule?: string }).rule }, violations: v }));
    process.exit(0);
  }
}
console.log(JSON.stringify({ found: false, seeds: SEEDS, triesPerSeed: TRIES, examined }));
