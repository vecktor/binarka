import { chromium } from '@playwright/test';
const port = process.argv[2];
const b = await chromium.launch();
const p = await (await b.newContext({ viewport: { width: 320, height: 700 } })).newPage();
await p.goto(`http://localhost:${port}/`); await p.locator('[data-cell]').first().waitFor();
const bad = [];
for (let w = 320; w <= 800; w++) {
  await p.setViewportSize({ width: w, height: 700 });
  const r = await p.evaluate(() => { const h = document.querySelector('header').getBoundingClientRect(); const kids = [...document.querySelector('header').children].map((c) => c.getBoundingClientRect()); const over = Math.max(...kids.map((k) => k.right)) - h.right; const wrap = new Set(kids.map((k) => Math.round(k.top + k.height / 2))).size > 1; return { over, wrap, scroll: document.documentElement.scrollWidth > innerWidth }; });
  if (r.over > 0.5 || r.scroll) bad.push(`${w}:${r.over.toFixed(1)}${r.scroll ? 'S' : ''}`);
}
console.log(bad.length ? `overflow at ${bad.length} widths: ${bad.join(' ')}` : 'no header overflow at any width 320..800 (1 px steps)');
await b.close();
